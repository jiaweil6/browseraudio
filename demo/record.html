03

Your voice as an instrument

A sampler in a dozen lines: every key plays your take, read faster for higher notes and slower for lower ones (it's treated as middle C). Click the keyboard and play a chord of yourself — or switch to Genie and let the genie sing.

python
import numpy as np
from browseraudio import keyboard

sr = 16000                                         # plenty for a voice, light to ship
take = rec.samples[:, 0]
take = np.interp(np.arange(0, len(take), rec.sample_rate / sr),   # resample to `sr`
                 np.arange(len(take)), take)
take = take / (np.abs(take).max() + 1e-9)          # normalize

def voice(freq):
    # Read the take faster for higher notes, slower for lower (it was "middle C").
    pos = np.arange(0, len(take) - 1, freq / 261.63)[: sr]   # at most 1 s per note
    i = pos.astype(int)
    y = take[i] + (pos - i) * (take[i + 1] - take[i])
    return 0.5 * y * np.minimum(1, np.arange(len(y)) / (0.01 * sr))  # 10 ms fade-in

kb = keyboard(voice, sr, release=0.15)
